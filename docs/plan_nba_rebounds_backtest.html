New export cell (add at end of notebook)

import boto3, io

S3_BUCKET = "nba-betting-mt"
BT_PREFIX = "data/04_output/backtest/rebounds"

PROD_SPEC    = "A3"
PROD_MODELS  = {"ols", "xgb"}
PROD_EDGE    = 0.05

def _upload_csv(df, s3_key):
    buf = io.BytesIO()
    df.to_csv(buf, index=False)
    buf.seek(0)
    boto3.client("s3").put_object(Bucket=S3_BUCKET, Key=s3_key, Body=buf.getvalue())
    print(f"Uploaded s3://{S3_BUCKET}/{s3_key}  ({len(df):,} rows)")

# --- game-level ---
multi = spec_sweep_pnl_df.loc[
    (spec_sweep_pnl_df["spec"]     == PROD_SPEC) &
    (spec_sweep_pnl_df["model"].isin(PROD_MODELS)) &
    (spec_sweep_pnl_df["min_edge"] == PROD_EDGE)
].copy()
_upload_csv(multi, f"{BT_PREFIX}/multi.csv")

# --- per-season summary ---
summary = (
    multi[multi["result"].isin({"win", "loss", "push"})]
    .groupby(["season", "model"], as_index=False)
    .agg(
        bets    = ("pnl_units", "count"),
        wins    = ("result",    lambda s: (s == "win").sum()),
        losses  = ("result",    lambda s: (s == "loss").sum()),
        pnl     = ("pnl_units", "sum"),
    )
)
summary["ats_pct"] = summary["wins"] / (summary["wins"] + summary["losses"])
_upload_csv(summary, f"{BT_PREFIX}/summary.csv")
